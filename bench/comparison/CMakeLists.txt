cmake_minimum_required(VERSION 3.20)
project(CrawComparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CRAW_ENCODER_DIR "${REPO_ROOT}/CinematicRAWStudio-Encoder" CACHE PATH "Encoder checkout")
set(CRAW_DECODER_DIR "${REPO_ROOT}/CinematicRAWStudio-Decoder" CACHE PATH "Decoder checkout")
set(LIBJXL_DIR "" CACHE PATH "Pinned libjxl source checkout with submodules")
if(NOT EXISTS "${LIBJXL_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set LIBJXL_DIR to a libjxl checkout with its submodules")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature TOOLS DEVTOOLS DOXYGEN MANPAGES BENCHMARK EXAMPLES JNI SJPEG OPENEXR VIEWERS PLUGINS TRANSCODE_JPEG)
  set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LIBJXL_DIR}" libjxl EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_executable(craw_compare compare.cpp
  "${CRAW_ENCODER_DIR}/src/Encoder.cpp"
  "${CRAW_DECODER_DIR}/src/Decoder.cpp"
  "${CRAW_DECODER_DIR}/src/Reader.cpp")
target_include_directories(craw_compare PRIVATE "${CRAW_ENCODER_DIR}/include" "${CRAW_DECODER_DIR}/include")
target_link_libraries(craw_compare PRIVATE jxl jxl_threads Threads::Threads)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  target_compile_options(craw_compare PRIVATE -march=armv8-a+crc)
endif()
